  <script>
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');

    // 画面サイズ・レスポンシブ設定 (16:9)
    const GAME_WIDTH = 800;
    const GAME_HEIGHT = 450;
    let scale = 1;

    function resizeCanvas() {
      const containerWidth = window.innerWidth;
      const containerHeight = window.innerHeight;
      
      let w = containerWidth;
      let h = containerWidth * (GAME_HEIGHT / GAME_WIDTH);

      if (h > containerHeight) {
        h = containerHeight;
        w = containerHeight * (GAME_WIDTH / GAME_HEIGHT);
      }

      canvas.width = GAME_WIDTH;
      canvas.height = GAME_HEIGHT;
      canvas.style.width = `${w}px`;
      canvas.style.height = `${h}px`;
      scale = w / GAME_WIDTH;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    // アクセスカウンター永続化
    let count = localStorage.getItem('taroman_counter') || 108;
    count = parseInt(count) + 1;
    localStorage.setItem('taroman_counter', count);
    document.getElementById('visitor-count').innerText = String(count).padStart(6, '0');

    // キー操作管理
    const keys = { left: false, right: false, up: false, down: false, punch: false, kick: false, beam: false };

    // タッチイベントバインド
    function bindTouch(elementId, keyName) {
      const el = document.getElementById(elementId);
      if (!el) return;
      el.addEventListener('touchstart', (e) => { e.preventDefault(); keys[keyName] = true; });
      el.addEventListener('touchend', (e) => { e.preventDefault(); keys[keyName] = false; });
      el.addEventListener('mousedown', () => { keys[keyName] = true; });
      el.addEventListener('mouseup', () => { keys[keyName] = false; });
    }
    bindTouch('btn-up', 'up');
    bindTouch('btn-down', 'down');
    bindTouch('btn-left', 'left');
    bindTouch('btn-right', 'right');
    bindTouch('btn-punch', 'punch');
    bindTouch('btn-kick', 'kick');
    bindTouch('btn-beam', 'beam');

    // キーボードバインド
    window.addEventListener('keydown', (e) => {
      if (e.key === 'a' || e.key === 'A' || e.key === 'ArrowLeft') keys.left = true;
      if (e.key === 'd' || e.key === 'D' || e.key === 'ArrowRight') keys.right = true;
      if (e.key === 'w' || e.key === 'W' || e.key === 'ArrowUp') keys.up = true;
      if (e.key === 's' || e.key === 'S' || e.key === 'ArrowDown') keys.down = true;
      if (e.key === 'j' || e.key === 'J') keys.punch = true;
      if (e.key === 'k' || e.key === 'K') keys.kick = true;
      if (e.key === 'l' || e.key === 'L') keys.beam = true;
    });
    window.addEventListener('keyup', (e) => {
      if (e.key === 'a' || e.key === 'A' || e.key === 'ArrowLeft') keys.left = false;
      if (e.key === 'd' || e.key === 'D' || e.key === 'ArrowRight') keys.right = false;
      if (e.key === 'w' || e.key === 'W' || e.key === 'ArrowUp') keys.up = false;
      if (e.key === 's' || e.key === 'S' || e.key === 'ArrowDown') keys.down = false;
      if (e.key === 'j' || e.key === 'J') keys.punch = false;
      if (e.key === 'k' || e.key === 'K') keys.kick = false;
      if (e.key === 'l' || e.key === 'L') keys.beam = false;
    });

    // ゲーム状態
    let gameState = 'TITLE'; // TITLE, SELECT_DIFF, PLAYING, GAMEOVER, CLEAR
    let difficulty = 'NORMAL'; // EASY, NORMAL, NANI_KORE
    let currentStage = 0;
    let isBuildingTaroman = false;

    // 奇獣データ（全9ステージ）
    const bossList = [
      { name: '森の掟', hp: 100, color: '#27ae60', type: 'MORI' },
      { name: '歓喜', hp: 120, color: '#f1c40f', type: 'KANKI' },
      { name: '縄文人', hp: 150, color: '#8e44ad', type: 'JOMON' },
      { name: '駄々っ子', hp: 110, color: '#e67e22', type: 'DADAKKO' },
      { name: '疾走する眼', hp: 130, color: '#e74c3c', type: 'EYE' },
      { name: '未来の顔', hp: 140, color: '#2980b9', type: 'MIRAI' },
      { name: '重工業巨人', hp: 180, color: '#34495e', type: 'BUILDING' },
      { name: 'あすの神話', hp: 200, color: '#c0392b', type: 'MYTH' },
      { name: '太陽の塔', hp: 300, color: '#f39c12', type: 'SUN' }
    ];

    // タローマン＆敵オブジェクト
    let player, enemy;
    let projectiles = [];
    let effects = [];

    class Player {
      constructor() {
        this.x = 100;
        this.y = 280;
        this.w = 50;
        this.h = 100;
        this.vx = 0;
        this.vy = 0;
        this.isGrounded = true;
        this.hp = 100;
        this.maxHp = 100;
        this.gauge = 0;
        this.state = 'IDLE'; // IDLE, PUNCH, KICK, BEAM, HURT, DOWN, STUN
        this.stateTimer = 0;
        this.stunTimer = 0;
        this.facing = 1;
      }

      update() {
        if (this.stunTimer > 0) {
          this.stunTimer--;
          return;
        }

        // 移動
        if (this.state === 'IDLE') {
          if (keys.left) this.vx = -4;
          else if (keys.right) this.vx = 4;
          else this.vx = 0;

          if (keys.up && this.isGrounded) {
            this.vy = -12;
            this.isGrounded = false;
          }

          // 攻撃
          if (keys.punch) {
            this.state = 'PUNCH';
            this.stateTimer = 15;
          } else if (keys.kick) {
            this.state = 'KICK';
            this.stateTimer = 18;
          } else if (keys.beam && this.gauge >= 100) {
            this.state = 'BEAM';
            this.stateTimer = 40;
            this.gauge = 0;
            // ビーム発射
            projectiles.push({
              x: this.x + 40,
              y: this.y + 25,
              w: 700,
              h: 30,
              life: 25,
              type: 'BEAM',
              owner: 'PLAYER'
            });
          }
        }

        // 重力演算
        this.vy += 0.6;
        this.x += this.vx;
        this.y += this.vy;

        if (this.y >= 280) {
          this.y = 280;
          this.vy = 0;
          this.isGrounded = true;
        }

        // ステートタイマー消費
        if (this.stateTimer > 0) {
          this.stateTimer--;
          if (this.stateTimer <= 0) this.state = 'IDLE';
        }

        // 画面外防止
        if (this.x < 10) this.x = 10;
        if (this.x > GAME_WIDTH - 60) this.x = GAME_WIDTH - 60;
      }

      draw() {
        ctx.save();
        ctx.translate(this.x + this.w / 2, this.y + this.h / 2);

        // ビル着用状態の描画 (重工業巨人クリア後)
        if (isBuildingTaroman) {
          ctx.fillStyle = '#2c3e50';
          ctx.fillRect(-35, -20, 70, 70);
          ctx.strokeStyle = '#fff';
          ctx.strokeRect(-35, -20, 70, 70);
        }

        // メラメラ炎の背景（顔の後ろ）
        for (let i = 0; i < 5; i++) {
          ctx.fillStyle = i % 2 === 0 ? '#ff0000' : '#f39c12';
          ctx.beginPath();
          ctx.arc(
            (Math.random() - 0.5) * 10,
            -50 + (Math.random() - 0.5) * 10,
            20 + Math.random() * 5,
            0, Math.PI * 2
          );
          ctx.fill();
        }

        // 首から上：金色
        ctx.fillStyle = '#f1c40f';
        ctx.beginPath();
        ctx.arc(0, -45, 18, 0, Math.PI * 2);
        ctx.fill();
        ctx.strokeStyle = '#d35400';
        ctx.lineWidth = 2;
        ctx.stroke();

        // 顔の模様
        ctx.fillStyle = '#000';
        ctx.fillRect(-6, -50, 4, 10);
        ctx.fillRect(2, -50, 4, 10);

        // 胴体
        if (!isBuildingTaroman) {
          ctx.fillStyle = '#e74c3c';
          ctx.fillRect(-15, -30, 30, 50);
          ctx.fillStyle = '#fff';
          ctx.fillRect(-5, -20, 10, 30);
        }

        // 足（1本蹴りのため、キック時は軸足1本のみ描画）
        ctx.fillStyle = '#bdc3c7';
        if (this.state === 'KICK') {
          // キック状態：軸足のみ
          ctx.fillRect(-10, 20, 10, 30); // 軸足
          // 蹴り足（前方に伸ばす）
          ctx.fillStyle = '#e74c3c';
          ctx.fillRect(0, -5, 45, 15);
        } else {
          // 通常状態：2本足
          ctx.fillRect(-12, 20, 10, 30);
          ctx.fillRect(2, 20, 10, 30);
        }

        // パンチ腕
        if (this.state === 'PUNCH') {
          ctx.fillStyle = '#e74c3c';
          ctx.fillRect(10, -25, 35, 12);
        } else if (this.state === 'BEAM') {
          // 手のひら突き出しポーズ
          ctx.fillStyle = '#e74c3c';
          ctx.fillRect(5, -20, 30, 10);
          ctx.fillStyle = '#f1c40f';
          ctx.beginPath();
          ctx.arc(35, -15, 8, 0, Math.PI * 2);
          ctx.fill();
        }

        // ビリビリ（麻痺）エフェクト
        if (this.stunTimer > 0) {
          ctx.strokeStyle = '#f1c40f';
          ctx.lineWidth = 3;
          ctx.beginPath();
          ctx.moveTo(-20, -50); ctx.lineTo(20, 40);
          ctx.moveTo(20, -50); ctx.lineTo(-20, 40);
          ctx.stroke();
        }

        ctx.restore();
      }
    }

    class Enemy {
      constructor(data) {
        this.name = data.name;
        this.type = data.type;
        this.hp = data.hp;
        this.maxHp = data.hp;
        this.x = 600;
        this.y = 280;
        this.w = 60;
        this.h = 100;
        this.vx = 0;
        this.vy = 0;
        this.color = data.color;
        this.state = 'IDLE'; // IDLE, ATTACK, DOWN, RISE
        this.stateTimer = 0;
        this.isGrounded = true;
        this.dadaCount = 0; // 駄々っ子自爆用

        // 太陽の塔は3倍サイズ
        if (this.type === 'SUN') {
          this.w = 120;
          this.h = 240;
          this.y = 140;
        }
      }

      update() {
        if (this.state === 'DOWN') {
          this.stateTimer--;
          if (this.stateTimer <= 0) {
            this.state = 'RISE';
            this.stateTimer = 20;
          }
          return;
        }

        if (this.state === 'RISE') {
          this.stateTimer--;
          if (this.stateTimer <= 0) this.state = 'IDLE';
          return;
        }

        // なんだこれモードでの高速ランダムアクション
        let speed = difficulty === 'EASY' ? 1 : difficulty === 'NORMAL' ? 2 : 5;

        if (this.state === 'IDLE') {
          // タローマンを追尾
          if (this.x > player.x + 70) this.x -= speed;
          else if (this.x < player.x - 70) this.x += speed;

          // ランダム特殊攻撃
          if (Math.random() < (difficulty === 'NANI_KORE' ? 0.08 : 0.02)) {
            this.attack();
          }
        }

        // 重力
        this.vy += 0.6;
        this.y += this.vy;

        let floor = this.type === 'SUN' ? 140 : 280;
        if (this.y >= floor) {
          this.y = floor;
          this.vy = 0;
          this.isGrounded = true;

          // 駄々っ子の激突自爆
          if (this.state === 'DADA_JUMP') {
            this.hp -= 35; // 3回でほぼ死亡
            this.state = 'DOWN';
            this.stateTimer = 30;
            effects.push({ x: this.x, y: this.y, text: '激突ドカン！', life: 30 });
          }
        }

        if (this.stateTimer > 0) {
          this.stateTimer--;
          if (this.stateTimer <= 0 && this.state !== 'DOWN') this.state = 'IDLE';
        }
      }

      attack() {
        this.state = 'ATTACK';
        this.stateTimer = 30;

        if (this.type === 'JOMON') {
          // 泥吐き
          projectiles.push({ x: this.x - 20, y: this.y + 20, vx: -5, vy: -2, radius: 12, type: 'MUD', owner: 'ENEMY' });
        } else if (this.type === 'MORI') {
          // 首コロコロ
          projectiles.push({ x: this.x, y: this.y + 20, vx: -6, vy: 0, radius: 18, type: 'HEAD', owner: 'ENEMY' });
        } else if (this.type === 'KANKI') {
          // 麻痺音符
          projectiles.push({ x: this.x - 20, y: this.y, vx: -4, vy: 0, radius: 10, type: 'NOTE', owner: 'ENEMY' });
        } else if (this.type === 'DADAKKO') {
          // 上空急降下
          this.vy = -16;
          this.state = 'DADA_JUMP';
        }
      }

      takeDamage(dmg) {
        if (this.state === 'DOWN') return;
        this.hp -= dmg;
        player.gauge = Math.min(100, player.gauge + 15);

        if (dmg >= 20) {
          this.state = 'DOWN';
          this.stateTimer = 40;
          this.vy = -6;
        }
      }

      draw() {
        ctx.save();
        ctx.translate(this.x + this.w / 2, this.y + this.h / 2);

        if (this.state === 'DOWN') {
          ctx.rotate(Math.PI / 2); // パタンと倒れる
        }

        // 種類ごとのグラフィック
        ctx.fillStyle = this.color;

        if (this.type === 'SUN') {
          // 太陽の塔（3倍サイズ）
          ctx.fillRect(-this.w/2, -this.h/2, this.w, this.h);
          ctx.fillStyle = '#f1c40f';
          ctx.beginPath();
          ctx.arc(0, -this.h/4, 30, 0, Math.PI*2); // 金の顔
          ctx.fill();
        } else if (this.type === 'JOMON') {
          // 縄文人
          ctx.fillRect(-25, -50, 50, 100);
          ctx.fillStyle = '#f39c12';
          ctx.beginPath(); ctx.arc(0, -35, 15, 0, Math.PI*2); ctx.fill();
        } else {
          // 一般奇獣
          ctx.fillRect(-this.w/2, -this.h/2, this.w, this.h);
        }

        ctx.restore();
      }
    }

    // ゲーム初期化
    function initGame() {
      player = new Player();
      enemy = new Enemy(bossList[currentStage]);
      projectiles = [];
      effects = [];
    }

    // 当たり判定・更新
    function update() {
      if (gameState !== 'PLAYING') return;

      player.update();
      enemy.update();

      // プレイヤーの攻撃ヒット判定
      if (player.state === 'PUNCH' && player.stateTimer === 10) {
        if (Math.abs(player.x - enemy.x) < 70) enemy.takeDamage(12);
      }
      if (player.state === 'KICK' && player.stateTimer === 12) {
        if (Math.abs(player.x - enemy.x) < 80) enemy.takeDamage(22);
      }

      // 弾の更新・判定
      for (let i = projectiles.length - 1; i >= 0; i--) {
        let p = projectiles[i];
        if (p.type === 'BEAM') {
          p.life--;
          if (enemy.state !== 'DOWN' && Math.abs(enemy.y - player.y) < 50) {
            enemy.takeDamage(2); // 連続ヒット
          }
          if (p.life <= 0) projectiles.splice(i, 1);
        } else {
          p.x += p.vx || 0;
          p.y += p.vy || 0;

          // プレイヤーへのヒット判定
          if (p.owner === 'ENEMY' && Math.abs(p.x - player.x) < 30 && Math.abs(p.y - player.y) < 50) {
            if (p.type === 'NOTE') {
              player.stunTimer = 90; // 1.5秒麻痺
              effects.push({ x: player.x, y: player.y, text: 'ビリビリ！', life: 30 });
            } else {
              player.hp -= 15;
            }
            projectiles.splice(i, 1);
          }
        }
      }

      // ステージクリア判定
      if (enemy.hp <= 0) {
        if (enemy.type === 'BUILDING') {
          isBuildingTaroman = true; // 重工業巨人着用形態へ！
        }

        currentStage++;
        if (currentStage >= bossList.length) {
          gameState = 'CLEAR';
        } else {
          enemy = new Enemy(bossList[currentStage]);
        }
      }

      // プレイヤー死亡
      if (player.hp <= 0) {
        gameState = 'GAMEOVER';
      }
    }

    // 描画
    function draw() {
      ctx.clearRect(0, 0, GAME_WIDTH, GAME_HEIGHT);

      // 背景
      ctx.fillStyle = '#111';
      ctx.fillRect(0, 0, GAME_WIDTH, GAME_HEIGHT);
      ctx.fillStyle = '#222';
      ctx.fillRect(0, 380, GAME_WIDTH, 70); // 地面

      if (gameState === 'TITLE') {
        ctx.fillStyle = '#e74c3c';
        ctx.font = 'bold 36px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('タローマン べらぼう格闘劇', GAME_WIDTH/2, 150);

        ctx.fillStyle = '#fff';
        ctx.font = '20px sans-serif';
        ctx.fillText('タップまたはクリックしてスタート！', GAME_WIDTH/2, 260);
        return;
      }

      if (gameState === 'SELECT_DIFF') {
        ctx.fillStyle = '#fff';
        ctx.font = '24px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('難易度を選択してください', GAME_WIDTH/2, 100);

        ctx.fillStyle = '#2ecc71';
        ctx.fillRect(250, 150, 300, 50);
        ctx.fillStyle = '#f39c12';
        ctx.fillRect(250, 220, 300, 50);
        ctx.fillStyle = '#e74c3c';
        ctx.fillRect(250, 290, 300, 50);

        ctx.fillStyle = '#fff';
        ctx.font = 'bold 20px sans-serif';
        ctx.fillText('超簡単', GAME_WIDTH/2, 182);
        ctx.fillText('普通', GAME_WIDTH/2, 252);
        ctx.fillText('なんだこれ (カオス)', GAME_WIDTH/2, 322);
        return;
      }

      if (gameState === 'PLAYING') {
        player.draw();
        enemy.draw();

        // 弾の描画
        projectiles.forEach(p => {
          if (p.type === 'BEAM') {
            ctx.fillStyle = `hsl(${Math.random() * 360}, 100%, 50%)`;
            ctx.fillRect(p.x, p.y - 15, p.w, p.h);
          } else if (p.type === 'MUD') {
            ctx.fillStyle = '#795548';
            ctx.beginPath(); ctx.arc(p.x, p.y, p.radius, 0, Math.PI*2); ctx.fill();
          } else if (p.type === 'NOTE') {
            ctx.fillStyle = '#f1c40f';
            ctx.font = '24px sans-serif';
            ctx.fillText('♪', p.x, p.y);
          } else {
            ctx.fillStyle = '#e74c3c';
            ctx.beginPath(); ctx.arc(p.x, p.y, p.radius || 10, 0, Math.PI*2); ctx.fill();
          }
        });

        // エフェクト描画
        effects.forEach(ef => {
          ctx.fillStyle = '#f1c40f';
          ctx.font = 'bold 18px sans-serif';
          ctx.fillText(ef.text, ef.x, ef.y);
          ef.life--;
        });

        // UI（体力・ゲージ）
        ctx.fillStyle = '#fff';
        ctx.font = 'bold 16px sans-serif';
        ctx.textAlign = 'left';
        ctx.fillText(`タローマン HP: ${player.hp}`, 20, 30);
        ctx.fillStyle = '#333';
        ctx.fillRect(20, 40, 200, 15);
        ctx.fillStyle = '#e74c3c';
        ctx.fillRect(20, 40, player.hp * 2, 15);

        // ビームゲージ
        ctx.fillStyle = '#fff';
        ctx.fillText(`芸術爆発ゲージ: ${player.gauge}%`, 20, 75);
        ctx.fillStyle = '#333';
        ctx.fillRect(20, 85, 200, 10);
        ctx.fillStyle = '#9b59b6';
        ctx.fillRect(20, 85, player.gauge * 2, 10);

        // 敵UI
        ctx.textAlign = 'right';
        ctx.fillText(`${enemy.name} HP: ${Math.max(0, enemy.hp)}`, GAME_WIDTH - 20, 30);
        ctx.fillStyle = '#333';
        ctx.fillRect(GAME_WIDTH - 220, 40, 200, 15);
        ctx.fillStyle = enemy.color;
        ctx.fillRect(GAME_WIDTH - 220, 40, Math.max(0, enemy.hp / enemy.maxHp * 200), 15);

        ctx.textAlign = 'center';
        ctx.fillText(`STAGE ${currentStage + 1} / 9`, GAME_WIDTH / 2, 30);
      }

      if (gameState === 'CLEAR') {
        ctx.fillStyle = '#f1c40f';
        ctx.font = 'bold 40px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('全奇獣 撃破！べらぼうな勝利！', GAME_WIDTH/2, 200);
      }

      if (gameState === 'GAMEOVER') {
        ctx.fillStyle = '#e74c3c';
        ctx.font = 'bold 40px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillText('爆発失敗... 芸術は爆発だ！', GAME_WIDTH/2, 200);
      }
    }

    // クリック判定（画面遷移用）
    canvas.addEventListener('click', (e) => {
      const rect = canvas.getBoundingClientRect();
      const clickX = (e.clientX - rect.left) / scale;
      const clickY = (e.clientY - rect.top) / scale;

      if (gameState === 'TITLE') {
        gameState = 'SELECT_DIFF';
      } else if (gameState === 'SELECT_DIFF') {
        if (clickX >= 250 && clickX <= 550) {
          if (clickY >= 150 && clickY <= 200) difficulty = 'EASY';
          if (clickY >= 220 && clickY <= 270) difficulty = 'NORMAL';
          if (clickY >= 290 && clickY <= 340) difficulty = 'NANI_KORE';
          currentStage = 0;
          initGame();
          gameState = 'PLAYING';
        }
      } else if (gameState === 'GAMEOVER' || gameState === 'CLEAR') {
        gameState = 'TITLE';
      }
    });

    // メインループ
    function gameLoop() {
      update();
      draw();
      requestAnimationFrame(gameLoop);
    }
    gameLoop();
  </script>
